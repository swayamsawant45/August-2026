use company_db;

-- sample setup so the queries below return results
drop table if exists Emp;
drop table if exists Dept;

create table Dept (
    DeptID int primary key,
    DeptName varchar(50),
    Location varchar(50)
);

create table Emp (
    EmpID int primary key,
    EmpName varchar(100),
    Salary decimal(10,2),
    DeptID int,
    JoiningDate date,
    foreign key (DeptID) references Dept(DeptID)
);

-- 1. total salary of all employees
select sum(Salary) as total_salary from Emp;

-- 2. average salary per department
select DeptID, avg(Salary) as avg_salary
from Emp
group by DeptID;

-- 3. employee count per department
select DeptID, count(*) as total_employees
from Emp
group by DeptID;

-- 4. departments with more than 5 employees
select DeptID, count(*) as total_employees
from Emp
group by DeptID
having count(*) > 5;

-- 5. distinct department locations
select distinct Location from Dept;

-- 6. highest salary and total number of employees
select max(Salary) as highest_salary from Emp;
select count(*) as total_employees from Emp;

-- 7. names starting with 'A'
select * from Emp where EmpName like 'A%';

-- 8. names ending with 'n'
select * from Emp where EmpName like '%n';

-- 9. names containing 'ra'
select * from Emp where EmpName like '%ra%';

-- 10. sorted by salary, highest first
select * from Emp order by Salary desc;

-- 11. DeptID ascending, then salary descending
select * from Emp order by DeptID asc, Salary desc;

-- 12. salary between 30000 and 60000
select * from Emp where Salary between 30000 and 60000;

-- 13. DeptID in (10, 20, 30)
select * from Emp where DeptID in (10, 20, 30);

-- 14. minimum salary
select min(Salary) as min_salary from Emp;

-- 15. joined between 2020-01-01 and 2021-12-31
select * from Emp
where JoiningDate between '2020-01-01' and '2021-12-31';

-- 16. salary is null
select * from Emp where Salary is null;

-- 17. salary is not null
select * from Emp where Salary is not null;

-- 18. departments where total salary is above 1,00,000
select DeptID, sum(Salary) as total_salary
from Emp
group by DeptID
having sum(Salary) > 100000;

-- 19. distinct employee names
select distinct EmpName from Emp;

-- 20. number of departments sharing the same location
select Location, count(*) as dept_count
from Dept
group by Location
having count(*) > 1;
