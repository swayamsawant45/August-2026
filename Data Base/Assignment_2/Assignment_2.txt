create database if not exists company_db;
use company_db;

create table Department (
    DeptID int primary key,
    DeptName varchar(50) not null,
    Location varchar(50)
);

create table Emp (
    EmpID int primary key,
    EmpName varchar(100) not null,
    Salary decimal(10,2) check (Salary > 0),
    DeptID int,
    JoiningDate date,
    foreign key (DeptID) references Department(DeptID)
);

-- privileges
create user if not exists 'readonly_user'@'localhost' identified by 'read123';
create user if not exists 'temp_user'@'localhost' identified by 'temp123';

grant select on company_db.Emp to 'readonly_user'@'localhost';

grant insert on company_db.Emp to 'temp_user'@'localhost';
revoke insert on company_db.Emp from 'temp_user'@'localhost';

-- insert a new employee and commit
start transaction;
insert into Emp (EmpID, EmpName, Salary, DeptID, JoiningDate, email)
values (101, 'Pooja Deshmukh', 52000, 20, '2023-02-14', 'pooja@example.com');
commit;

-- update salary of EmpID = 101 and commit
start transaction;
update Emp set Salary = 58000 where EmpID = 101;
commit;

-- savepoint before updating DeptID, then rollback to it
start transaction;
savepoint before_dept_update;
update Emp set DeptID = 30 where EmpID = 101;
rollback to before_dept_update;
commit;

-- employees with salary above 50000
select * from Emp where Salary > 50000;

-- 10% raise for DeptID = 10
update Emp set Salary = Salary * 1.10 where DeptID = 10;

-- employee count per department
select DeptID, count(*) as total_employees
from Emp
group by DeptID;

-- departments with no employees
select d.DeptID, d.DeptName
from Department d
left join Emp e on d.DeptID = e.DeptID
where e.EmpID is null;

-- highest salary
select max(Salary) as highest_salary from Emp;

-- 2nd highest salary

from Emp
